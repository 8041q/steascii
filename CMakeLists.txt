cmake_minimum_required(VERSION 3.20)
project(Steascii VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Include the C++ runtime in the binary; users need no redistributable installer.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

function(steascii_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(steascii_text src/text.cpp)
target_include_directories(steascii_text PUBLIC src)
steascii_warnings(steascii_text)

include(CTest)
if(BUILD_TESTING)
    add_executable(text_tests tests/text_tests.cpp)
    target_link_libraries(text_tests PRIVATE steascii_text)
    steascii_warnings(text_tests)
    add_test(NAME text_conversion COMMAND text_tests)
endif()

if(WIN32)
    enable_language(RC)
    add_library(steascii_windows src/windows_support.cpp)
    target_link_libraries(steascii_windows PUBLIC steascii_text user32)
    target_compile_definitions(steascii_windows PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    steascii_warnings(steascii_windows)

    configure_file(src/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    add_executable(Steascii WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
    target_link_libraries(Steascii PRIVATE steascii_windows comdlg32 comctl32 gdi32)
    steascii_warnings(Steascii)
    if(MSVC)
        # The resource file embeds our privilege/DPI manifest.
        target_link_options(Steascii PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_link_options(Steascii PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()

    if(BUILD_TESTING)
        add_executable(windows_tests tests/windows_tests.cpp)
        target_link_libraries(windows_tests PRIVATE steascii_windows)
        steascii_warnings(windows_tests)
        if(MINGW)
            target_link_options(text_tests PRIVATE -static -static-libgcc -static-libstdc++)
            target_link_options(windows_tests PRIVATE -static -static-libgcc -static-libstdc++)
        endif()
        add_test(NAME windows_file_and_clipboard COMMAND windows_tests)
    endif()
else()
    message(STATUS "Building portable text tests only. The desktop app requires Windows.")
endif()
